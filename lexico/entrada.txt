clase @ClaseP {

var &x, &y, &idx, $nom, %prom;
var &vec[10], &mat[3];

metodo vacio @main() {

    // Lecturas iniciales
    leer(&x, &y, $nom);

    // Asignaciones simples
    &x = 10;
    &y = &x + 5 * (3 - 1);
    %prom = 15.5;

    // Arreglos
    &idx = 5;
    &vec[&idx] = &x + 3;
    &mat[3] = 100;

    // Escribir salida inicial
    escribir(&x, &y, %prom, $nom);

    // Estructura SI
    si(&x > &y) {
        escribir(&x, $nom);
    } sino {
        escribir(&y, %prom);
    }

    // MIENTRAS
    mientras(&x < 20) {
        &x = &x + 1;
        escribir(&x);
    }

    // REPITE
    repite {
        &y = &y - 1;
    } &y > 5;

    // SWITCH
    switch(&x) {
        encaso 10:
            escribir(&x);
        encaso 20:
            escribir($nom);
    }

    // Llamada a método
    ejecutar @proc;

    // Regresar sin expresión
    regresar;

    // Salida del método
    salir;
}

metodo vacio @proc(&val, $eti) {

    escribir(&val, $eti);

    si(&val == 0) {
        escribir($eti);
    }

    regresar;
}

}
